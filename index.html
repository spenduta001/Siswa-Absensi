// ============================================================================
// BACKEND GOOGLE APPS SCRIPT - ABSENSI DIGITAL UPTD SMPN 2 KETAPANG
// 7 Tab Otomatis: Data_Pengguna, Data_Siswa, Data_Kelas_Guru, Absensi_Masuk,
// Absensi_Pulang, Pengaturan_Sistem, Log_Aktivitas
// ============================================================================

const SHEET_SCHEMAS = {
  'Data_Pengguna': ['username', 'password', 'name', 'role', 'roleLabel'],
  'Data_Siswa': ['id', 'nis', 'nisn', 'nama', 'jk', 'kelas', 'ortu', 'wa', 'alamat'],
  'Data_Kelas_Guru': ['name', 'level', 'wali'],
  'Absensi_Masuk': ['id', 'date', 'nisn', 'nama', 'kelas', 'jam_masuk', 'status', 'wa_status'],
  'Absensi_Pulang': ['id', 'date', 'nisn', 'nama', 'kelas', 'jam_pulang', 'wa_status'],
  'Pengaturan_Sistem': ['key', 'value'],
  'Log_Aktivitas': ['waktu', 'aksi', 'detail', 'status']
};

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  initSheets(ss);

  // Jika dipanggil dengan parameter ?action=pull, kembalikan JSON database
  if (e && e.parameter && e.parameter.action === 'pull') {
    return jsonResponse(pullAllData());
  }

  // Jika dibuka langsung di browser sebagai Web App, tampilkan file HTML 'Index'
  try {
    return HtmlService.createHtmlOutputFromFile('Index')
      .setTitle('SISTEM INFORMASI ABSENSI DIGITAL SISWA - UPTD SMPN 2 KETAPANG')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
      .addMetaTag('viewport', 'width=device-width, initial-scale=1');
  } catch (err) {
    return jsonResponse({ status: 'ok', message: 'API Backend Aktif. Buat file HTML bernama Index untuk antarmuka langsung.' });
  }
}

function doPost(e) {
  try {
    // Mendukung pengiriman dari Hidden Form (e.parameter.payload) maupun Fetch JSON (e.postData.contents)
    const raw = (e && e.parameter && e.parameter.payload) ? e.parameter.payload : (e && e.postData ? e.postData.contents : '{}');
    const payload = JSON.parse(raw);
    return jsonResponse(processSyncPayload(payload));
  } catch (err) {
    return jsonResponse({ status: 'error', message: err.toString() });
  }
}

// Dipanggil langsung oleh google.script.run dari browser
function serverSyncAction(payload) {
  return processSyncPayload(payload);
}

function pullAllData() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  initSheets(ss);

  const masukList = sheetToObjects(ss.getSheetByName('Absensi_Masuk'));
  const pulangList = sheetToObjects(ss.getSheetByName('Absensi_Pulang'));
  const pulangMap = {};
  pulangList.forEach(p => { pulangMap[p.date + '_' + p.nisn] = p.jam_pulang; });

  const mergedAttendance = masukList.map(m => ({
    id: m.id,
    date: m.date,
    nisn: m.nisn,
    nama: m.nama,
    kelas: m.kelas,
    jam_masuk: m.jam_masuk,
    jam_pulang: pulangMap[m.date + '_' + m.nisn] || '',
    status: m.status,
    wa_status: m.wa_status
  }));

  return {
    status: 'success',
    users: sheetToObjects(ss.getSheetByName('Data_Pengguna')),
    students: sheetToObjects(ss.getSheetByName('Data_Siswa')),
    classes: sheetToObjects(ss.getSheetByName('Data_Kelas_Guru')),
    attendance: mergedAttendance
  };
}

function processSyncPayload(payload) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  initSheets(ss);
  const action = payload.action;

  if (action === 'record_attendance') {
    const d = payload.data;
    if (d.jam_masuk && d.jam_masuk !== '-') {
      upsertRow(ss.getSheetByName('Absensi_Masuk'), 'id', [
        d.id, d.date, d.nisn, d.nama, d.kelas, d.jam_masuk, d.status, d.wa_status || 'Terkirim'
      ]);
    }
    if (d.jam_pulang && d.jam_pulang !== '-') {
      upsertRow(ss.getSheetByName('Absensi_Pulang'), 'id', [
        'PLG-' + d.date + '-' + d.nisn, d.date, d.nisn, d.nama, d.kelas, d.jam_pulang, d.wa_status || 'Terkirim'
      ]);
    }
    logActivity(ss, 'SCAN_' + (d.jam_pulang ? 'PULANG' : 'MASUK'), d.nama + ' (' + d.kelas + ') - ' + d.status, 'OK');

    if (payload.wa && payload.wa.endpoint && payload.wa.token && payload.wa.phone) {
      sendWhatsAppGateway(payload.wa, ss);
    }
    return { status: 'success', message: 'Absensi tersimpan ke Google Sheets' };
  }

  if (action === 'sync_all') {
    if (payload.users) overwriteSheet(ss.getSheetByName('Data_Pengguna'), SHEET_SCHEMAS['Data_Pengguna'], payload.users);
    if (payload.students) overwriteSheet(ss.getSheetByName('Data_Siswa'), SHEET_SCHEMAS['Data_Siswa'], payload.students);
    if (payload.classes) overwriteSheet(ss.getSheetByName('Data_Kelas_Guru'), SHEET_SCHEMAS['Data_Kelas_Guru'], payload.classes);
    if (payload.attendance) {
      const masukRows = payload.attendance.map(a => ({
        id: a.id, date: a.date, nisn: a.nisn, nama: a.nama, kelas: a.kelas,
        jam_masuk: a.jam_masuk || '-', status: a.status, wa_status: a.wa_status || '-'
      }));
      const pulangRows = payload.attendance.filter(a => a.jam_pulang && a.jam_pulang !== '-').map(a => ({
        id: 'PLG-' + a.date + '-' + a.nisn, date: a.date, nisn: a.nisn, nama: a.nama,
        kelas: a.kelas, jam_pulang: a.jam_pulang, wa_status: a.wa_status || 'Terkirim'
      }));
      overwriteSheet(ss.getSheetByName('Absensi_Masuk'), SHEET_SCHEMAS['Absensi_Masuk'], masukRows);
      overwriteSheet(ss.getSheetByName('Absensi_Pulang'), SHEET_SCHEMAS['Absensi_Pulang'], pulangRows);
    }
    if (payload.settings) {
      const setRows = Object.keys(payload.settings).map(k => ({ key: k, value: String(payload.settings[k]) }));
      overwriteSheet(ss.getSheetByName('Pengaturan_Sistem'), SHEET_SCHEMAS['Pengaturan_Sistem'], setRows);
    }
    logActivity(ss, 'SYNC_ALL', 'Sinkronisasi penuh seluruh tabel', 'OK');
    return { status: 'success', message: 'Sinkronisasi 7 tab berhasil' };
  }

  if (action === 'send_wa') {
    const ok = sendWhatsAppGateway(payload.wa, ss);
    return { status: ok ? 'success' : 'error' };
  }

  return { status: 'unknown_action' };
}

function sendWhatsAppGateway(wa, ss) {
  try {
    const response = UrlFetchApp.fetch(wa.endpoint, {
      method: 'post',
      headers: { 'Authorization': wa.token },
      payload: { target: wa.phone, message: wa.message },
      muteHttpExceptions: true
    });
    logActivity(ss, 'WA_NOTIF', (wa.studentName || wa.phone) + ' -> ' + wa.phone, 'HTTP ' + response.getResponseCode());
    return true;
  } catch (e) {
    logActivity(ss, 'WA_ERROR', wa.phone, e.toString());
    return false;
  }
}

function logActivity(ss, aksi, detail, status) {
  const sh = ss.getSheetByName('Log_Aktivitas');
  if (sh) sh.appendRow([new Date().toLocaleString('id-ID', { timeZone: 'Asia/Jakarta' }), aksi, detail, status]);
}

function initSheets(ss) {
  Object.keys(SHEET_SCHEMAS).forEach(name => {
    let sh = ss.getSheetByName(name);
    if (!sh) {
      sh = ss.insertSheet(name);
      sh.appendRow(SHEET_SCHEMAS[name]);
      sh.getRange(1, 1, 1, SHEET_SCHEMAS[name].length).setFontWeight('bold').setBackground('#0F294A').setFontColor('#FFFFFF');
    }
  });
}

function sheetToObjects(sheet) {
  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return [];
  const headers = data[0];
  return data.slice(1).map(row => {
    const obj = {};
    headers.forEach((h, i) => obj[h] = String(row[i]));
    return obj;
  });
}

function upsertRow(sheet, keyCol, rowValues) {
  const data = sheet.getDataRange().getValues();
  const keyIdx = data[0].indexOf(keyCol);
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][keyIdx]) === String(rowValues[keyIdx])) {
      sheet.getRange(i + 1, 1, 1, rowValues.length).setValues([rowValues]);
      return;
    }
  }
  sheet.appendRow(rowValues);
}

function overwriteSheet(sheet, headers, items) {
  sheet.clearContents();
  sheet.appendRow(headers);
  if (items && items.length > 0) {
    const rows = items.map(item => headers.map(h => item[h] || ''));
    sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
  }
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
